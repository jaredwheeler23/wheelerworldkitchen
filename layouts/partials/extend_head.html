{{- /* Universal Recipe Schema Markup - Applies to ALL recipes with rating parameter */ -}} {{- if and (.IsPage) (.Params.rating) -}} {{- $content := .RawContent -}} {{- $ingredientsSection := "" -}} {{- $instructionsSection := "" -}} {{- /* Extract Ingredients section */ -}} {{- $ingredientsStart := index (findRE `(?s)## Ingredients\s*\n` $content) 0 -}} {{- if $ingredientsStart -}} {{- $afterIngredients := replaceRE `(?s)^.*?## Ingredients\s*\n` "" $content -}} {{- $ingredientsSection = index (findRE `(?s)^.*?(?=\n## )` $afterIngredients) 0 -}} {{- end -}} {{- /* Extract Instructions section */ -}} {{- $instructionsStart := index (findRE `(?s)## Instructions\s*\n` $content) 0 -}} {{- if $instructionsStart -}} {{- $afterInstructions := replaceRE `(?s)^.*?## Instructions\s*\n` "" $content -}} {{- $instructionsSection = index (findRE `(?s)^.*?(?=\n## |\z)` $afterInstructions) 0 -}} {{- end -}} {{- /* Parse ingredients into array */ -}} {{- $ingredients := slice -}} {{- range split $ingredientsSection "\n" -}} {{- $line := trim . " \t\r" -}} {{- if and (hasPrefix $line "- ") (gt (len $line) 2) -}} {{- $ingredient := trim (substr $line 2) " " -}} {{- $ingredients = $ingredients | append $ingredient -}} {{- end -}} {{- end -}} {{- /* Parse instructions into array */ -}} {{- $instructions := slice -}} {{- range split $instructionsSection "\n" -}} {{- $line := trim . " \t\r" -}} {{- if findRE `^\d+\.\s+` $line -}} {{- $instruction := trim (replaceRE `^\d+\.\s+` "" $line) " " -}} {{- if gt (len $instruction) 0 -}} {{- $instructions = $instructions | append $instruction -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}}
