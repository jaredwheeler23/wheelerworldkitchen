{{- /* Universal Recipe Schema Markup - Applies to ALL recipes with rating parameter */ -}} {{- if and (.IsPage) (.Params.rating) -}} {{- $content := .RawContent -}} {{- $ingredientsSection := "" -}} {{- $instructionsSection := "" -}} {{- /* Extract Ingredients section */ -}} {{- if strings.Contains $content "## Ingredients" -}} {{- $parts := split $content "## Ingredients" -}} {{- if gt (len $parts) 1 -}} {{- $afterIngredients := index $parts 1 -}} {{- $nextSection := split $afterIngredients "## " -}} {{- $ingredientsSection = index $nextSection 0 -}} {{- end -}} {{- end -}} {{- /* Extract Instructions section */ -}} {{- if strings.Contains $content "## Instructions" -}} {{- $parts := split $content "## Instructions" -}} {{- if gt (len $parts) 1 -}} {{- $afterInstructions := index $parts 1 -}} {{- $nextSection := split $afterInstructions "## " -}} {{- $instructionsSection = index $nextSection 0 -}} {{- end -}} {{- end -}} {{- /* Parse ingredients into array */ -}} {{- $ingredients := slice -}} {{- range split $ingredientsSection "\n" -}} {{- $line := trim . " \t\r" -}} {{- if and (hasPrefix $line "- ") (gt (len $line) 2) -}} {{- $ingredient := trim (substr $line 2) " " -}} {{- $ingredients = $ingredients | append $ingredient -}} {{- end -}} {{- end -}} {{- /* Parse instructions into array */ -}} {{- $instructions := slice -}} {{- $stepPattern := `^\d+\.\s+` -}} {{- range split $instructionsSection "\n" -}} {{- $line := trim . " \t\r" -}} {{- if findRE $stepPattern $line -}} {{- $instruction := trim (replaceRE $stepPattern "" $line) " " -}} {{- if gt (len $instruction) 0 -}} {{- $instructions = $instructions | append $instruction -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}}
